# Butterfly meadow: a bee, two frames (wing up / wing flat). 80 ms/frame.
NewDocument(10, 8)
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#FFC02EFF)
Stroke([(2,3),(8,7)])
SelectTool(Pencil); SetBrushSize(1); SetPrimaryColor(#201810FF)
Stroke([(4,3),(4,7)])
Stroke([(6,3),(6,7)])
Tap(8,4)
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#DAE8F0AA)
Stroke([(3,0),(6,2)])
DuplicateFrame(0)
SetActiveFrame(1)
SelectTool(SelectRect); SetSelectionMode(Replace)
Stroke([(0,0),(9,2)])
ClearSelection()
SelectNone()
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#DAE8F0AA)
Stroke([(4,1),(8,2)])
SetActiveFrame(0)
SetAllDurations(80)
