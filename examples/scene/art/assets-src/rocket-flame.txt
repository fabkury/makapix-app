# Rocket launch: the exhaust flame, three flickering frames. 40 ms/frame = 1 scene frame.
NewDocument(14, 22)
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#F06018FF)
Stroke([(2,0),(11,12)])
SetPrimaryColor(#FFB03CFF)
Stroke([(4,0),(9,10)])
SetPrimaryColor(#FFF0A0FF)
Stroke([(5,0),(8,6)])
SelectTool(Pencil); SetBrushSize(2); SetPrimaryColor(#F06018FF)
Stroke([(6,12),(6,19)])
SetBrushSize(1); SetPrimaryColor(#FFB03CFF)
Stroke([(6,11),(6,15)])
DuplicateFrame(0)
SetActiveFrame(1)
SelectAll()
ClearSelection()
SelectNone()
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#F06018FF)
Stroke([(3,0),(10,10)])
SetPrimaryColor(#FFB03CFF)
Stroke([(4,0),(9,8)])
SetPrimaryColor(#FFF0A0FF)
Stroke([(5,0),(8,5)])
SelectTool(Pencil); SetBrushSize(2); SetPrimaryColor(#F06018FF)
Stroke([(6,10),(7,15)])
DuplicateFrame(1)
SetActiveFrame(2)
SelectAll()
ClearSelection()
SelectNone()
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#F06018FF)
Stroke([(2,0),(11,11)])
SetPrimaryColor(#FFB03CFF)
Stroke([(3,0),(10,9)])
SetPrimaryColor(#FFF0A0FF)
Stroke([(5,0),(8,6)])
SelectTool(Pencil); SetBrushSize(2); SetPrimaryColor(#F06018FF)
Stroke([(7,11),(5,17)])
SetBrushSize(1); SetPrimaryColor(#FFB03CFF)
Stroke([(6,10),(7,13)])
SetActiveFrame(0)
SetAllDurations(40)
