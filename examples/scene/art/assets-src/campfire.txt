# Campfire night: crossed logs + a four-frame flame cycle (the flame region is cleared and
# redrawn per frame while the logs persist). 80 ms/frame = 2 scene frames at 25 fps.
NewDocument(28, 30)
# logs
SelectTool(Line); SetBrushSize(3); SetPrimaryColor(#4A2E14FF)
Stroke([(3,27),(24,23)])
SetPrimaryColor(#5C3A1AFF)
Stroke([(3,23),(24,27)])
# frame 0 flame
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#E8641EFF)
Stroke([(8,8),(19,23)])
SetPrimaryColor(#FFA23CFF)
Stroke([(10,12),(17,23)])
SetPrimaryColor(#FFE28CFF)
Stroke([(12,16),(15,22)])
SelectTool(Pencil); SetBrushSize(1); SetPrimaryColor(#E8641EFF)
Stroke([(13,7),(12,4)])
# frame 1: taller, leaning left
DuplicateFrame(0)
SetActiveFrame(1)
SelectTool(SelectRect); SetSelectionMode(Replace)
Stroke([(0,0),(27,21)])
ClearSelection()
SelectNone()
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#E8641EFF)
Stroke([(9,6),(18,23)])
SetPrimaryColor(#FFA23CFF)
Stroke([(11,10),(16,23)])
SetPrimaryColor(#FFE28CFF)
Stroke([(12,15),(15,22)])
SelectTool(Pencil); SetBrushSize(1); SetPrimaryColor(#E8641EFF)
Stroke([(11,5),(10,2)])
# frame 2: wider, squat
DuplicateFrame(1)
SetActiveFrame(2)
SelectTool(SelectRect); SetSelectionMode(Replace)
Stroke([(0,0),(27,21)])
ClearSelection()
SelectNone()
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#E8641EFF)
Stroke([(7,10),(20,23)])
SetPrimaryColor(#FFA23CFF)
Stroke([(9,13),(18,23)])
SetPrimaryColor(#FFE28CFF)
Stroke([(12,17),(16,22)])
SelectTool(Pencil); SetBrushSize(1); SetPrimaryColor(#FFA23CFF)
Stroke([(15,9),(16,7)])
# frame 3: tall, leaning right
DuplicateFrame(2)
SetActiveFrame(3)
SelectTool(SelectRect); SetSelectionMode(Replace)
Stroke([(0,0),(27,21)])
ClearSelection()
SelectNone()
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#E8641EFF)
Stroke([(9,7),(18,23)])
SetPrimaryColor(#FFA23CFF)
Stroke([(11,11),(17,23)])
SetPrimaryColor(#FFE28CFF)
Stroke([(13,16),(16,22)])
SelectTool(Pencil); SetBrushSize(1); SetPrimaryColor(#E8641EFF)
Stroke([(16,6),(17,3)])
SetActiveFrame(0)
SetAllDurations(80)
