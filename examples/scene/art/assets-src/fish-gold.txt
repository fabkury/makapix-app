# Aquarium: a right-facing goldfish, two frames (tail level / tail flicked up).
# 120 ms/frame = 3 scene frames at 25 fps.
NewDocument(26, 16)
# body + belly + fin + eye
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#F09030FF)
Stroke([(6,3),(23,12)])
SetPrimaryColor(#FFD489FF)
Stroke([(9,8),(22,12)])
SetPrimaryColor(#D87820FF)
Stroke([(12,2),(17,6)])
SelectTool(Pencil); SetBrushSize(1); SetPrimaryColor(#141414FF)
Tap(20,6)
SetPrimaryColor(#FFFFFFFF)
Tap(20,5)
# frame-0 tail: a level wedge
SetPrimaryColor(#E8862CFF)
Stroke([(6,5),(3,5)])
Stroke([(6,6),(2,6)])
Stroke([(6,7),(1,7)])
Stroke([(6,8),(1,8)])
Stroke([(6,9),(2,9)])
Stroke([(6,10),(3,10)])
# frame 1: tail flicked upward
DuplicateFrame(0)
SetActiveFrame(1)
SelectTool(SelectRect); SetSelectionMode(Replace)
Stroke([(0,0),(5,15)])
ClearSelection()
SelectNone()
SelectTool(Pencil); SetBrushSize(1); SetPrimaryColor(#E8862CFF)
Stroke([(6,4),(3,1)])
Stroke([(6,5),(2,2)])
Stroke([(6,6),(1,3)])
Stroke([(6,7),(1,4)])
Stroke([(6,8),(2,5)])
Stroke([(6,9),(3,6)])
SetActiveFrame(0)
SetAllDurations(120)
