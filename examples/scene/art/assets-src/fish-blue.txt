# Aquarium: a small right-facing teal fish, two frames. 120 ms/frame.
NewDocument(18, 11)
SelectTool(Ellipse); SetShapeFill(true); SetPrimaryColor(#38C8C0FF)
Stroke([(4,2),(15,8)])
SetPrimaryColor(#9AE8E0FF)
Stroke([(6,5),(14,8)])
SelectTool(Pencil); SetBrushSize(1); SetPrimaryColor(#101010FF)
Tap(13,4)
SetPrimaryColor(#2AA49CFF)
Stroke([(4,3),(1,3)])
Stroke([(4,4),(0,4)])
Stroke([(4,5),(0,5)])
Stroke([(4,6),(1,6)])
DuplicateFrame(0)
SetActiveFrame(1)
SelectTool(SelectRect); SetSelectionMode(Replace)
Stroke([(0,0),(3,10)])
ClearSelection()
SelectNone()
SelectTool(Pencil); SetBrushSize(1); SetPrimaryColor(#2AA49CFF)
Stroke([(4,3),(1,0)])
Stroke([(4,4),(0,1)])
Stroke([(4,5),(0,2)])
Stroke([(4,6),(1,3)])
SetActiveFrame(0)
SetAllDurations(120)
