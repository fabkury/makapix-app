# Sailing day: a gull, two frames (wings down / wings up). 160 ms/frame = 4 scene frames.
NewDocument(14, 10)
SelectTool(Pencil); SetBrushSize(1); SetPrimaryColor(#2E3640FF)
Stroke([(0,6),(4,3),(6,5)])
Stroke([(7,5),(9,3),(13,6)])
SetPrimaryColor(#4A525CFF)
Tap(6,6); Tap(7,6)
DuplicateFrame(0)
SetActiveFrame(1)
SelectAll()
ClearSelection()
SelectNone()
SelectTool(Pencil); SetBrushSize(1); SetPrimaryColor(#2E3640FF)
Stroke([(0,2),(4,5),(6,5)])
Stroke([(7,5),(9,5),(13,2)])
SetPrimaryColor(#4A525CFF)
Tap(6,6); Tap(7,6)
SetActiveFrame(0)
SetAllDurations(160)
