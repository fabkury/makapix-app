# Sailing day — a little sloop bobbing between three parallax wave bands (each scrolls a
# whole number of scallop periods per loop, so the wrap is seamless), gulls on their own
# flap clocks, drifting clouds. 128x128 @ 25 fps, 4-second loop. Positions are prop centers.
NewScene(128,128,25000)
SetFrameCount(100)
SetBackground(#175E90FF)
@import(assets/sky-backdrop.mkpx, Sky and sea)
@import(assets/sun.mkpx, Sun)
@import(assets/cloud.mkpx, Cloud)
@import(assets/wave-strip.mkpx, Waves)
@import(assets/boat.mkpx, Sloop)
@import(assets/gull.mkpx, Gull)
PlaceActorAt(1, 64, 64)
PlaceActorAt(2, 99, 21)
# clouds (actors 3-4), full-loop crossers (off-canvas at both ends, so the wrap hides)
PlaceActorAt(3, -40, 24)
SetKeyTween(3, x, 0, -40, linear)
SetKey(3, x, 99, 150)
PlaceActorAt(3, 150, 38)
SetBase(4, scale, 800)
SetKeyTween(4, x, 0, 150, linear)
SetKey(4, x, 99, -30)
# far wave band (actor 5): one period per loop
PlaceActorAt(4, 112, 81)
SetBase(5, opacity, 210)
SetKeyTween(5, x, 0, 112, linear)
SetKey(5, x, 99, 80)
# mid band (actor 6): two periods
PlaceActorAt(4, 112, 93)
SetKeyTween(6, x, 0, 112, linear)
SetKey(6, x, 99, 48)
# the sloop (actor 7): bobbing and rolling between mid and near water
PlaceActorAt(5, 64, 73)
SetKeyTween(7, y, 0, 73, easeinout)
SetKeyTween(7, y, 24, 76, easeinout)
SetKeyTween(7, y, 49, 73, easeinout)
SetKeyTween(7, y, 74, 76, easeinout)
SetKey(7, y, 99, 73)
SetKeyTween(7, rot, 0, 0, easeinout)
SetKeyTween(7, rot, 12, 3000, easeinout)
SetKeyTween(7, rot, 37, -3000, easeinout)
SetKeyTween(7, rot, 62, 3000, easeinout)
SetKeyTween(7, rot, 87, -3000, easeinout)
SetKey(7, rot, 99, 0)
# near band (actor 8): three periods, in front of the hull
PlaceActorAt(4, 112, 111)
SetKeyTween(8, x, 0, 112, linear)
SetKey(8, x, 99, 16)
# gulls (actors 9-10)
PlaceActorAt(6, -16, 26)
SetKeyTween(9, x, 0, -16, linear)
SetKey(9, x, 60, 140)
SetKeyTween(9, y, 0, 26, easeinout)
SetKeyTween(9, y, 30, 20, easeinout)
SetKey(9, y, 60, 28)
PlaceActorAt(6, 140, 36)
SetBase(10, fliph, 1)
SetKeyTween(10, x, 0, 140, hold)
SetKeyTween(10, x, 25, 140, linear)
SetKey(10, x, 90, -16)
SetKeyTween(10, y, 25, 36, easeinout)
SetKeyTween(10, y, 58, 30, easeinout)
SetKey(10, y, 90, 38)
SetLoop(0, 99)
SetPlayhead(0)
SelectNone()
