# Testkit: a scale pulse (0.5×–2.4×, cleanEdge), hold-key flip toggles on asymmetric
# art, and a style comparison — two identical noise props at 30°, cleanEdge vs nearest.
# Build: mkpx scene run examples/scene/testkit/scale-and-flips.txt save:scale-and-flips.mkps
NewScene(96,64,20000)
SetFrameCount(40)
SetBackground(#12151AFF)
GenProp(20,20,1,3)
SetPropStyle(1, cleanedge)
GenProp(16,16,1,9)
GenProp(12,12,1,5)
SetPropStyle(3, cleanedge)
GenProp(12,12,1,5)
# Pulse: breathe about the center.
PlaceActorAt(1, 24, 26)
RenameActor(1, Pulse)
SetBase(1, pivotx, 10000)
SetBase(1, pivoty, 10000)
SetKeyTween(1, scale, 0, 1000, easeinout)
SetKeyTween(1, scale, 20, 2400, easeinout)
SetKey(1, scale, 39, 1000)
# Flipper: H flips every 5 frames (hold-only track), one V flip mid-loop.
PlaceActorAt(2, 64, 16)
RenameActor(2, Flipper)
SetBase(2, pivotx, 8000)
SetBase(2, pivoty, 8000)
SetKeyTween(2, fliph, 0, 0, hold)
SetKeyTween(2, fliph, 5, 1, hold)
SetKeyTween(2, fliph, 10, 0, hold)
SetKeyTween(2, fliph, 15, 1, hold)
SetKeyTween(2, fliph, 20, 0, hold)
SetKeyTween(2, fliph, 25, 1, hold)
SetKeyTween(2, fliph, 30, 0, hold)
SetKeyTween(2, fliph, 35, 1, hold)
SetKeyTween(2, flipv, 0, 0, hold)
SetKeyTween(2, flipv, 20, 1, hold)
# StyleTwins: same seed, same 30° — crisp vs chunky side by side.
PlaceActorAt(3, 30, 50)
RenameActor(3, CleanEdge)
SetBase(3, pivotx, 6000)
SetBase(3, pivoty, 6000)
SetBase(3, rot, 30000)
PlaceActorAt(4, 66, 50)
RenameActor(4, Nearest)
SetBase(4, pivotx, 6000)
SetBase(4, pivoty, 6000)
SetBase(4, rot, 30000)
SetLoop(0, 39)
SetPlayhead(0)
