# Testkit: three rotation behaviors — a center-pivot spin (cleanEdge), a clock hand
# ticking on hold keys, and a moon orbiting via a pivot placed OUTSIDE its prop.
# Build: mkpx scene run examples/scene/testkit/spin-orbit-tick.txt save:spin-orbit-tick.mkps
NewScene(96,96,25000)
SetFrameCount(100)
SetBackground(#0E1116FF)
GenProp(24,24,1,7)
SetPropStyle(1, cleanedge)
GenPropSolid(4,28,#E0E0E0FF)
GenPropSolid(8,8,#FFC94DFF)
GenPropSolid(10,10,#4DE8D2FF)
# Spinner: pivot at the prop center, one full turn per loop (3.6°/frame, seamless wrap).
PlaceActorAt(1, 26, 64)
RenameActor(1, Spinner)
SetBase(1, pivotx, 12000)
SetBase(1, pivoty, 12000)
SetKeyTween(1, rot, 0, 0, linear)
SetKey(1, rot, 99, 356400)
# Ticker: pivot at the hand's base, 30° hold steps — a clock, not a sweep.
PlaceActorAt(2, 68, 64)
RenameActor(2, Ticker)
SetBase(2, pivotx, 2000)
SetBase(2, pivoty, 26000)
SetKeyTween(2, rot, 0, 0, hold)
SetKeyTween(2, rot, 8, 30000, hold)
SetKeyTween(2, rot, 16, 60000, hold)
SetKeyTween(2, rot, 24, 90000, hold)
SetKeyTween(2, rot, 32, 120000, hold)
SetKeyTween(2, rot, 40, 150000, hold)
SetKeyTween(2, rot, 48, 180000, hold)
SetKeyTween(2, rot, 56, 210000, hold)
SetKeyTween(2, rot, 64, 240000, hold)
SetKeyTween(2, rot, 72, 270000, hold)
SetKeyTween(2, rot, 80, 300000, hold)
SetKeyTween(2, rot, 88, 330000, hold)
# Planet + orbiting moon: the moon's pivot sits 18 px left of its art, so rotating
# swings the art around the pivot point — an orbit with a single rot track.
PlaceActorAt(4, 43, 25)
RenameActor(3, Planet)
PlaceActorAt(3, 48, 30)
RenameActor(4, Moon)
SetBase(4, pivotx, -18000)
SetBase(4, pivoty, 4000)
SetKeyTween(4, rot, 0, 0, linear)
SetKey(4, rot, 99, -356400)
SetLoop(0, 99)
SetPlayhead(0)
