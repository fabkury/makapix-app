# Testkit: the fastest curated rate (50 fps, exact 2 cs/frame). A glider rides a
# ring of linear keys while spinning two turns — motion should look liquid-smooth
# in playback AND in the exported files.
# Build: mkpx scene run examples/scene/testkit/fifty-fps-glide.txt save:fifty-fps-glide.mkps
NewScene(96,96,50000)
SetFrameCount(100)
SetBackground(#0C0F14FF)
GenPropSolid(6,6,#3080E0FF)
GenProp(14,14,1,17)
SetPropStyle(2, cleanedge)
PlaceActorAt(1, 45, 45)
RenameActor(1, Hub)
PlaceActorAt(2, 78, 48)
RenameActor(2, Glider)
SetBase(2, pivotx, 7000)
SetBase(2, pivoty, 7000)
SetKeyTween(2, x, 0, 78, linear)
SetKeyTween(2, x, 12, 69, linear)
SetKeyTween(2, x, 25, 48, linear)
SetKeyTween(2, x, 37, 27, linear)
SetKeyTween(2, x, 50, 18, linear)
SetKeyTween(2, x, 62, 27, linear)
SetKeyTween(2, x, 75, 48, linear)
SetKeyTween(2, x, 87, 69, linear)
SetKey(2, x, 99, 78)
SetKeyTween(2, y, 0, 48, linear)
SetKeyTween(2, y, 12, 27, linear)
SetKeyTween(2, y, 25, 18, linear)
SetKeyTween(2, y, 37, 27, linear)
SetKeyTween(2, y, 50, 48, linear)
SetKeyTween(2, y, 62, 69, linear)
SetKeyTween(2, y, 75, 78, linear)
SetKeyTween(2, y, 87, 69, linear)
SetKey(2, y, 99, 50)
SetKeyTween(2, rot, 0, 0, linear)
SetKey(2, rot, 99, 712800)
SetLoop(0, 99)
SetPlayhead(0)
